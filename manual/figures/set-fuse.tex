\begin{document}
\begin{tikzpicture}[ads]

    \node[index] at ($(f-0.north west)+(\s*29.6mm+14.8mm,-15mm)$) {segment \s};
  }
  \draw[line width=1.4pt,draw=adsInk] ($(f-19.north east)+(0,1.5mm)$) -- ($(f-19.south east)+(0,-1.5mm)$);
  \node[var] (k) at ($(f-11.north)+(0,20mm)$) {$F[ h_0 ] \oplus F[ h_1 ] \oplus F[ h_2 ] = f( h )$};
