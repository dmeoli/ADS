% node. Build: make figures (in ../build).

% \lrow{name}{y}{x of the first node}{item/style list}{arc style list}
% draws a row of nodes 14 mm apart, the arcs between them (the k-th arc
% style is that of the arc out of node k) and nil after the last node
\newcommand{\lrow}[5]{%
  \foreach \v/\s [count=\i from 0] in {#4}
    \node[\s] (#1-\i) at (#3+\i*14mm,#2) {\v};
  \foreach \s [count=\i from 0] in {#5} {
    \pgfmathtruncatemacro{\j}{\i+1}
    \draw[\s] (#1-\i) -- (#1-\j);}}
