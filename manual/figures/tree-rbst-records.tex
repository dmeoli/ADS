% The node j, first of i, ..., floor(x) in the order of insertion: i goes
% left of j, the search path for x right (Morin, figure 7.3). Build: make figures (in ../build).
\documentclass[border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-ads}
\begin{document}
\begin{tikzpicture}[ads]
  \node[vertex-touched] (j) at (0,0) {$j$};
  \node[node,minimum width=34mm,font=\small] (L) at (-24mm,-14mm) {$\ldots , i , \ldots , j - 1$};
  \node[node,minimum width=34mm,font=\small] (R) at (24mm,-14mm) {$j + 1 , \ldots , \lfloor x \rfloor , \ldots$};
  \draw[edge] (j) -- (L.north); \draw[edge-touched] (j) -- (R.north);
  \node[node,minimum width=22mm,font=\small] (A) at (0,13mm) {the path above $j$};
  \draw[edge-touched] (A) -- (j);
  \node[var,anchor=west] at (34mm,-3mm) {search path for $x$};
\end{tikzpicture}
\end{document}
