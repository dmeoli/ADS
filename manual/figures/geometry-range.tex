% Source for the figure of RangeTree (sec. 16.4): seven points in the map
% ordered by (x, y), each node with its point (above) and the sorted
% ordinates of its subtree (below); above, the pieces into which count(2,
% 5, 2, 5) cuts the keys from (2, -inf) to (5, +inf) (shaded); below, the
% tree after insert(5, 6), the new node and the sequences merged again
% shaded. Build: make figures (in ../build).

\input{tikz-ads}
\tikzset{
  rp/.style={rectangle split,rectangle split parts=2,draw=white,
             line width=1.2pt,rounded corners=1.6pt,
             font=\small\ttfamily,text=adsInk,inner xsep=1.5mm,
             inner ysep=1.0mm,align=center,
             rectangle split part fill={adsTint,adsTint}},
  rpt/.style={rp,rectangle split part fill={adsPrimary,adsPrimary},
              text=adsOnPrimary},
  rpk/.style={rp,rectangle split part fill={adsPrimary,adsTint}},
  rph/.style={rp,rectangle split part fill={adsTint,adsAdded}},
  rpa/.style={rp,rectangle split part fill={adsAdded,adsAdded},
              text=adsOnAdded}
}
