% remainder 77 that a query of (2, 77) finds. Build: make figures (in
% ../build).

\begin{document}
\begin{tikzpicture}[ads]

  \foreach \i in {0,...,9} \node[index,below=1.6mm of h-\i] {\i};
  \node[var,anchor=east] at (-6mm,-16mm) {body};
  \cells{b}{0,-16mm}{12/cell,40/cell,3/touched,77/touched,200/touched,9/cell}
  \foreach \i in {0,...,5} \node[index,below=1.6mm of b-\i] {\i};
  \foreach \q/\x in {0/0.5,1/2,2/5,3/8} \node[index] at (\x*7mm,6mm) {$q = \q$};
  \node[index,anchor=west] at (46mm,-16mm) {the run of $q = 2$: slots $[ z_1 + 1 - 2 , z_2 - 2 ) = [ 2 , 5 )$};
