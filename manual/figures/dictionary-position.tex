% Source for the figure of PositionIndex (sec. 7.6): the pairs (address,
% position) of eight objects in the order of the places, before and after
% renumber(E) with E = {2, 5}, and the bit vector B of E between them.
% Build: make figures (in ../build).
\documentclass[border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-ads}
\tikzset{pc/.style={cell,minimum width=8mm,minimum height=6mm},
         pt/.style={touched,minimum width=8mm,minimum height=6mm},
         pa/.style={added,minimum width=8mm,minimum height=6mm},
         pf/.style={free,minimum width=8mm,minimum height=6mm}}
\begin{document}
\begin{tikzpicture}[ads]
  \node[var] at (4mm,8mm) {before};
  \foreach \o/\p/\s [count=\i from 0] in {o5/5/pt,o0/0/pc,o3/3/pc,o7/7/pc,o2/2/pt,o6/6/pc,o1/1/pc,o4/4/pc} {
    \node[pc] (ka-\i) at (0,-\i*7mm) {$o_{\p}$};
    \node[\s] (va-\i) at (8.4mm,-\i*7mm) {\p};}
  \node[var] at (40mm,8mm) {$B$, $E = \{ 2 , 5 \}$};
  \foreach \b [count=\i from 0] in {0,0,1,0,0,1,0,0} {
    \ifnum\b=1 \node[touched,minimum width=6mm] (b-\i) at ($(28mm,-14mm)+(\i*6.4mm,0)$) {\b};
    \else \node[cell,minimum width=6mm] (b-\i) at ($(28mm,-14mm)+(\i*6.4mm,0)$) {\b}; \fi
    \node[index,above=1.6mm of b-\i] {\the\numexpr\i-1+1\relax};}
  \node[var,anchor=north] at ($(b-3.south)!0.5!(b-4.south)+(0,-3mm)$) {$p \gets p - \mathrm{rank}( p )$};
  \node[var] at (88mm,8mm) {\texttt{renumber(E)}};
  \foreach \o/\p/\s [count=\i from 0] in {5/{}/pf,0/0/pc,3/2/pa,7/5/pa,2/{}/pf,6/4/pa,1/1/pc,4/3/pa} {
    \ifx\p\empty \node[pf] (kb-\i) at (84mm,-\i*7mm) {};
    \else \node[pc] (kb-\i) at (84mm,-\i*7mm) {$o_{\o}$}; \fi
    \node[\s] (vb-\i) at (92.4mm,-\i*7mm) {\p};}
  \node[index,anchor=west] at ($(vb-0.east)+(1mm,0)$) {$o_5$ removed};
  \node[index,anchor=west] at ($(vb-4.east)+(1mm,0)$) {$o_2$ removed};
\end{tikzpicture}
\end{document}
