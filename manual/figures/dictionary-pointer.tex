\begin{tikzpicture}[ads]

  \node[index,anchor=east] at ($(p-0.west)+(-2mm,0)$) {pointer};

    \node[var] at ($(h-\i.north)+(0,7.5mm)$) {\k};
  \node[var,anchor=west] at (62mm,0) {$h = \mathtt{4B}$, first place 5};
  \node[var,anchor=west] at (62mm,-7.4mm) {probes 5, 6, 7, 0, then 1};
