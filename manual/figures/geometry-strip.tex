% Source for the figure of the strip of closestPair() (sec. 16.3): the
% split x = x_m, the strip of width 2 delta about it, and the rectangle of
% delta x 2 delta above a point p of the strip, cut into the 8 squares of
% side delta / 2 that hold at most one point each. Build: make figures (in
% ../build).
\documentclass[border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-ads}
\tikzset{
  pt/.style={vertex-touched,minimum size=2mm},
  far/.style={vertex,minimum size=1.8mm}
}
\begin{document}
\begin{tikzpicture}[ads,x=8mm,y=8mm]
  % the strip and the rectangle above p
  \fill[adsFree] (-2,-0.9) rectangle (2,5.4);
  \fill[adsTint] (-2,1.4) rectangle (2,3.4);
  \foreach \x in {-1,0,1} \draw[white,line width=1pt] (\x,1.4) -- (\x,3.4);
  \draw[white,line width=1pt] (-2,2.4) -- (2,2.4);
  
  \draw[edge,dashed] (-2,-0.9) -- (-2,5.4);
  \draw[edge,dashed] (2,-0.9) -- (2,5.4);
  \draw[edge-touched] (0,-1.0) -- (0,5.6);
  \node[var,anchor=south] at (0,5.7) {$x = x_m$};
  % the points: in the strip, and out of it on either side, those of a
  % side at least delta apart
  \node[pt] (p) at (0.7,1.4) {};
  \node[font=\small,text=adsInk,anchor=north west] at (0.75,1.33) {$p$};
  \foreach \x/\y in {-1.5/-0.2,-0.5/1.9,-1.6/4.0,1.6/3.3,0.5/5.0,1.4/-0.6}
    \node[pt] at (\x,\y) {};
  \foreach \x/\y in {-3.2/1.0,-3.6/4.4,3.2/0.3,3.4/2.4,2.9/4.9}
    \node[far] at (\x,\y) {};
  % the measures
  \draw[<->,edge] (-2,-1.35) -- node[fill=white,inner sep=1pt] {$\delta$} (0,-1.35);
  \draw[<->,edge] (0,-1.35) -- node[fill=white,inner sep=1pt] {$\delta$} (2,-1.35);
  \draw[<->,edge] (2.3,1.4) -- node[fill=white,inner sep=1pt] {$\delta$} (2.3,3.4);
  \draw[<->,edge] (-2.3,2.4) -- node[fill=white,inner sep=1pt,left=0.5mm]
    {$\delta / 2$} (-2.3,3.4);
\end{tikzpicture}
\end{document}
