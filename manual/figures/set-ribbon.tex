% Source for the figure of RibbonFilter (sec. 10.10): the system in
% echelon form after the banding, m = 10 rows, windows of w = 4
% coefficients; a row holds an equation, its window of 4 coefficients
% written from its column, the first one (always 1) in violet, or none
% (faintest tint), and f is the right side. Build: make
% figures (in ../build).
\documentclass[border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{xstring}
\usepackage{lamalunga}
\input{tikz-ads}
\begin{document}
\begin{tikzpicture}[ads,
    b/.style={cell,minimum width=4.5mm,minimum height=4.5mm,rounded corners=0.8pt,font=\scriptsize},
    bt/.style={touched,minimum width=4.5mm,minimum height=4.5mm,rounded corners=0.8pt,font=\scriptsize},
    bf/.style={free,minimum width=4.5mm,minimum height=4.5mm,rounded corners=0.8pt}]
  \foreach \r/\bits in {0/1011,1/1101,2/,3/1110,4/1001,5/1011,6/,7/1100,8/,9/1000} {
    \node[index,anchor=east] at (-3.5mm,-\r*4.5mm) {\r};
    \ifx\bits\empty
      \foreach \c in {0,...,9} \node[bf] at (\c*4.5mm,-\r*4.5mm) {};
      \node[bf] at (51mm,-\r*4.5mm) {};
    \else
      \foreach \c in {0,...,9} \node[b] at (\c*4.5mm,-\r*4.5mm) {};
      \foreach \j in {0,1,2,3} {
        \pgfmathtruncatemacro{\c}{\r+\j}
        \StrChar{\bits}{\numexpr\j+1\relax}[\d]
        \ifnum\c<10
          \ifnum\j=0
            \node[bt] at (\c*4.5mm,-\r*4.5mm) {\d};
          \else
            \node[b] at (\c*4.5mm,-\r*4.5mm) {\d};
          \fi
        \fi }
      \node[b] at (51mm,-\r*4.5mm) {};
    \fi
  }
  \node[index] at (51mm,5mm) {$f$};
  \node[index] at (20.25mm,5mm) {columns};
\end{tikzpicture}
\end{document}
