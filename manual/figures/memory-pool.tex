% Source for the figure of ObjectPool (sec. 5.5): the blocks of 1, 2 and 4
% slots, each after its header H, which links it to the block before; the
% slots are numbered from 0 under the cells, and a free slot holds the
% number of the next free slot (- for nil). The free list F before and after
% free(e). Build: make figures (in ../build).
\documentclass[border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-ads}
% \pool{name}{y}{s2}{s4}: the three blocks of a row, with the text/style of
% the slots 2 and 4
\newcommand{\pool}[4]{%
  \cells{#1a}{0,#2}{H/cell,-/free}
  \cells{#1b}{24mm,#2}{H/cell,b/cell,#3}
  \cells{#1c}{55mm,#2}{H/cell,d/cell,#4,f/cell,/free}
  \foreach \s/\n in {#1a-1/0,#1b-1/1,#1b-2/2,#1c-1/3,#1c-2/4,#1c-3/5,#1c-4/6}
    \node[index,below=1.2mm of \s] {\n};}
\begin{document}
\begin{tikzpicture}[ads]
  \node[var,anchor=east] at (-6mm,0) {before};
  \pool{u}{0}{0/free}{e/cell}
  \draw[ptr] (uc-0.north) to[bend right=25] (ub-0.north);
  \draw[ptr] (ub-0.north) to[bend right=25] (ua-0.north);
  \node[var] (last) at (62mm,13mm) {$\mathit{last}$};
  \draw[ptr] (last) -- (uc-0.north east);
  \node[var] (n1) at (90mm,13mm) {$\mathit{next}$};
  \draw[ptr] (n1) -- (uc-4.north);
  \node[var,anchor=west] at (95mm,0) {$F = 2$};
  \node[var,anchor=east] at (-6mm,-20mm) {after};
  \pool{v}{-20mm}{0/free}{2/touched}
  \node[var,anchor=west] at (95mm,-20mm) {$F = 4$};
  \draw[ptr] (69mm,-12mm) -- (vc-2.north);
  \node[font=\scriptsize\ttfamily,text=adsInk,anchor=west] at (70mm,-12.5mm) {free(e)};
\end{tikzpicture}
\end{document}
