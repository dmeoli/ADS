\begin{tikzpicture}[ads]
  \node[on] (r) at (0,0) {$m = 10$\\$x_0$};

  \node[box] (b) at (25mm,-18mm) {$m = 2$\\$x_4$};
  \node[box] (c) at (-42mm,-36mm) {$m = 4$\\$x_2$};
