% \blk{name}{x}{y}{item/style list}: a block of places 6 mm apart;
% \nblock{name} marks a new block with a bar under it
\newcommand{\blk}[4]{%
  \foreach \v/\s [count=\i from 0] in {#4}
    \node[\s] (#1-\i) at (#2+\i*6mm,#3) {\v};}
\newcommand{\dotsat}[2]{\node[nullptr] at (#1,#2) {\dots};}
\newcommand{\nblock}[1]{%
  \draw[edge-added] ($(#1-0.south west)+(0.4mm,-1.2mm)$) --
    node[index,below=1.2mm] {new block}
    ($(#1-3.south east)+(-0.4mm,-1.2mm)$);}
